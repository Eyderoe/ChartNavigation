cmake_minimum_required(VERSION 3.28)
project(ChartNaviMSFS LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
include(CTest)

add_library(msfs_client STATIC SimConnectClient.cpp SimConnectClient.hpp SimConnectApi.hpp)
target_link_libraries(msfs_client PUBLIC Qt6::Core)
target_include_directories(msfs_client PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(msfs_monitor main.cpp)
target_link_libraries(msfs_monitor PRIVATE msfs_client)
include("${CMAKE_CURRENT_SOURCE_DIR}/deployRuntime.cmake")
msfs_deploy_runtime(msfs_monitor)

if(BUILD_TESTING AND WIN32)
    add_library(mock_simconnect SHARED test/mock_simconnect.cpp)
    target_include_directories(mock_simconnect PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    set_target_properties(mock_simconnect PROPERTIES PREFIX "")
    add_executable(msfs_tests test/client_test.cpp ../../src/connector/msfs.cpp)
    target_link_libraries(msfs_tests PRIVATE msfs_client)
    target_include_directories(msfs_tests PRIVATE ../../include ../../src)
    add_test(NAME MSFSSimConnect COMMAND msfs_tests $<TARGET_FILE:mock_simconnect>)
endif()
